เชื่อมต่อฐานข้อมูล Supabase

กรอกครั้งเดียว ระบบจะจำไว้ในเบราว์เซอร์นี้

แนะนำ: แก้ที่ไฟล์เดียว ใช้ได้ทุกเครื่อง

เปิดไฟล์ javascript/supabase-config.js (อยู่ในโฟลเดอร์ javascript ของเว็บ) แล้วใส่ Project URL กับ Anon Key ตรงนั้นแทน จะไม่ต้องมากรอกซ้ำทุกครั้งที่เปิดเครื่องใหม่หรือเบราว์เซอร์ใหม่ เพราะเว็บจะดึงค่าจากไฟล์นี้อัตโนมัติ (ไฟล์เดียวกับที่ index.html และ project.html ใช้อยู่แล้วด้วย)

— หรือกรอกที่นี่ชั่วคราว (จะจำไว้แค่ในเบราว์เซอร์นี้เครื่องเดียว) —

หาได้ที่ Supabase Dashboard > Settings > API

ยังไม่เคยตั้งตารางในฐานข้อมูล? กดดูคำสั่ง SQL 
create table projects (
  id text primary key,
  category text,
  badge_bg text,
  badge_text text,
  card_tag_color text,
  title text,
  location text,
  image text,
  placeholder text,
  client text,
  facts jsonb default '[]',
  gallery jsonb default '[]',
  sort_order int default 0,
  created_at timestamp default now()
);

alter table projects enable row level security;

create policy "public can read"
  on projects for select using (true);

create policy "public can write"
  on projects for all using (true) with check (true);

แล้วไปที่ Storage > New bucket ตั้งชื่อ project-images เลือก Public bucket
